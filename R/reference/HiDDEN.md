## Arguments

- x:

  input object

- ...:

  extra arguments passed into pipeline

- cond_col:

  character character indicating per-cell metadata column for condition
  labels

- control_val:

  value in per-cell condition labels representing control condition

- algo:

  Pipeline

## Value

HiDDEN output - list of two (2) elements w/ names:

- `p_hat`: HiDDEN pipeline per-cell score values

- `labs`: HiDDEN-adjusted per-cell condition labels

if (FALSE) { # requireNamespace("irlba", quietly = TRUE) && requireNamespace("Seurat", quietly = TRUE) && requireNamespace("SeuratObject", quietly = TRUE) && requireNamespace("SeuratData", quietly = TRUE) && ("ifnb" %in% SeuratData::InstalledData()[["Dataset"]])

  Seurat::SCTransform(verbose = FALSE) |>
  suppressWarnings() |>
  suppressMessages()
sct_pca <- function(so, k) {
  irlba::prcomp_irlba(
    t(SeuratObject::LayerData(so, assay = "SCT", layer = "scale.data")),
    n = k
  )$x
}
algo <- adapters$Pipeline(to_step(sct_pca), methods$reg_logit, methods$bin_kmeans)
out <- HiDDEN(so, "stim", "CTRL", algo, so = so, k = 15L)
names(out)
}
